# --- 1. FIRST TIME SETUP ---
git config --global user.name "Your Name"
git config --global user.email "your.email@example.com"

# --- 2. CREATE A NEW LOCAL REPO & PUSH TO GITHUB ---
git init
git add .
git commit -m "Initial commit"
git branch -M main
git remote add origin https://github.com
git push -u origin main

# --- 3. COPY AN EXISTING REPO ---
git clone https://github.com

# --- 4. DAILY WORKFLOW (SAVE & SYNC) ---
git status
git add .
git commit -m "Describe your changes here"
git pull origin main
git push origin main

# --- 5. BRANCHING & MERGING ---
git checkout -b feature-branch
git add .
git commit -m "Finished feature"
git checkout main
git merge feature-branch
git branch -d feature-branch

# --- 6. UNDO CHANGES ---
git restore filename.js       # Discard unstaged changes in a file
git reset HEAD~1             # Undo last commit, keep your code changes
git log --oneline            # View simple commit history
